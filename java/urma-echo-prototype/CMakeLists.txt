cmake_minimum_required(VERSION 3.16)
project(brpc_java_urma_echo LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This prototype targets Linux; do not infer Linux/URMA validation from a host build.")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(ECHO_ENABLE_URMA "Allow URMA mode (real runtime path must still be verified)" ON)
set(BRPC_ROOT "$ENV{BRPC_ROOT}" CACHE PATH "Existing bRPC source/build root or install prefix")
set(BRPC_EXTRA_INCLUDE_DIRS "" CACHE STRING "Explicit extra matching bRPC generated include directories")

find_package(Threads REQUIRED)
find_package(Protobuf CONFIG QUIET)
if(NOT TARGET protobuf::libprotobuf)
    find_package(Protobuf REQUIRED)
endif()
find_package(Java 1.8 REQUIRED COMPONENTS Development Runtime)
get_filename_component(_javac_real "${Java_JAVAC_EXECUTABLE}" REALPATH)
get_filename_component(_java_bin "${_javac_real}" DIRECTORY)
get_filename_component(_java_home "${_java_bin}" DIRECTORY)
find_path(JAVA_JNI_INCLUDE jni.h HINTS "$ENV{JAVA_HOME}" "${_java_home}" PATH_SUFFIXES include)
find_path(JAVA_JNI_PLATFORM jni_md.h HINTS "${JAVA_JNI_INCLUDE}" PATH_SUFFIXES linux)
if(NOT JAVA_JNI_INCLUDE OR NOT JAVA_JNI_PLATFORM)
    message(FATAL_ERROR "JDK development headers missing: require jni.h and include/linux/jni_md.h; set JAVA_HOME to the JDK.")
endif()

find_path(BRPC_INCLUDE_DIR brpc/server.h
    HINTS "${BRPC_ROOT}/output/include" "${BRPC_ROOT}/include" "${BRPC_ROOT}/build/output/include" "${BRPC_ROOT}/src")
set(_old_suffixes ${CMAKE_FIND_LIBRARY_SUFFIXES})
set(CMAKE_FIND_LIBRARY_SUFFIXES .so)
find_library(BRPC_LIBRARY NAMES brpc
    HINTS "${BRPC_ROOT}/output/lib" "${BRPC_ROOT}/output/lib64" "${BRPC_ROOT}/lib" "${BRPC_ROOT}/lib64"
          "${BRPC_ROOT}/build/output/lib" "${BRPC_ROOT}/build/lib" "${BRPC_ROOT}/build/src" "${BRPC_ROOT}/build/src/brpc")
set(CMAKE_FIND_LIBRARY_SUFFIXES ${_old_suffixes})
if(NOT BRPC_INCLUDE_DIR OR NOT BRPC_LIBRARY)
    message(FATAL_ERROR "Matching bRPC headers/libbrpc.so not found. Supply BRPC_ROOT or BRPC_INCLUDE_DIR and BRPC_LIBRARY. Existing bRPC will not be rebuilt.")
endif()
if(NOT BRPC_LIBRARY MATCHES "\\.so(\\.[0-9]+)*$")
    message(FATAL_ERROR "This first build supports existing shared libbrpc.so only. Static/PIC/transitive linkage needs separate review.")
endif()
find_path(GFLAGS_INCLUDE_DIR gflags/gflags.h)
find_library(GFLAGS_LIBRARY NAMES gflags)
if(NOT GFLAGS_INCLUDE_DIR OR NOT GFLAGS_LIBRARY)
    message(FATAL_ERROR "gflags development headers/library missing")
endif()
add_library(brpc_existing SHARED IMPORTED)
set_target_properties(brpc_existing PROPERTIES IMPORTED_LOCATION "${BRPC_LIBRARY}")
get_filename_component(_brpc_lib_dir "${BRPC_LIBRARY}" DIRECTORY)
file(SHA256 "${BRPC_LIBRARY}" _brpc_sha)
file(GLOB_RECURSE _fingerprint_sources CONFIGURE_DEPENDS
    "${CMAKE_SOURCE_DIR}/native/*" "${CMAKE_SOURCE_DIR}/java/*" "${CMAKE_SOURCE_DIR}/proto/*")
file(SHA256 "${CMAKE_SOURCE_DIR}/CMakeLists.txt" _cmake_sha)
set(_fingerprint "${_brpc_sha};${_cmake_sha};${CMAKE_CXX_COMPILER_ID};${CMAKE_CXX_COMPILER_VERSION};${CMAKE_SYSTEM_PROCESSOR};${CMAKE_BUILD_TYPE};${CMAKE_CXX_FLAGS};${CMAKE_CXX_FLAGS_RELEASE};${Protobuf_VERSION};${protobuf_VERSION};${ECHO_ENABLE_URMA}")
foreach(_source IN LISTS _fingerprint_sources)
    file(SHA256 "${_source}" _source_sha)
    string(APPEND _fingerprint ";${_source_sha}")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_fingerprint_sources})
string(SHA256 ECHO_BUILD_ID "${_fingerprint}")
file(WRITE "${CMAKE_BINARY_DIR}/build-id.txt" "${ECHO_BUILD_ID}\n")

set(_cpp_standard 14)
if(Protobuf_VERSION VERSION_GREATER "4.21" OR protobuf_VERSION VERSION_GREATER "4.21")
    set(_cpp_standard 17)
endif()
if(TARGET protobuf::protoc)
    set(_protoc "$<TARGET_FILE:protobuf::protoc>")
else()
    if(NOT Protobuf_PROTOC_EXECUTABLE)
        find_program(Protobuf_PROTOC_EXECUTABLE protoc)
    endif()
    if(NOT Protobuf_PROTOC_EXECUTABLE)
        message(FATAL_ERROR "protoc is required; it must match the linked Protobuf")
    endif()
    set(_protoc "${Protobuf_PROTOC_EXECUTABLE}")
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated" "${CMAKE_BINARY_DIR}/classes" "${CMAKE_BINARY_DIR}/generated-jni")
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/generated/echo.pb.cc" "${CMAKE_BINARY_DIR}/generated/echo.pb.h"
    COMMAND ${_protoc} "--cpp_out=${CMAKE_BINARY_DIR}/generated" "-I${CMAKE_SOURCE_DIR}/proto" "${CMAKE_SOURCE_DIR}/proto/echo.proto"
    DEPENDS "${CMAKE_SOURCE_DIR}/proto/echo.proto" VERBATIM)

add_library(echo_core STATIC native/core.cpp "${CMAKE_BINARY_DIR}/generated/echo.pb.cc")
target_compile_features(echo_core PUBLIC cxx_std_${_cpp_standard})
target_compile_definitions(echo_core PUBLIC ECHO_ENABLE_URMA=$<BOOL:${ECHO_ENABLE_URMA}> ECHO_BUILD_ID="${ECHO_BUILD_ID}")
target_compile_options(echo_core PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_include_directories(echo_core PUBLIC native "${CMAKE_BINARY_DIR}/generated" "${BRPC_INCLUDE_DIR}"
    "${GFLAGS_INCLUDE_DIR}" ${BRPC_EXTRA_INCLUDE_DIRS})
target_link_libraries(echo_core PUBLIC brpc_existing protobuf::libprotobuf "${GFLAGS_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})

file(GLOB JAVA_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/java/prototype/echo/*.java")
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/java.stamp" "${CMAKE_BINARY_DIR}/generated-jni/prototype_echo_NativeBridge.h"
    COMMAND "${Java_JAVAC_EXECUTABLE}" -encoding UTF-8 -source 8 -target 8
            -d "${CMAKE_BINARY_DIR}/classes" -h "${CMAKE_BINARY_DIR}/generated-jni" ${JAVA_SOURCES}
    COMMAND "${CMAKE_COMMAND}" -E touch "${CMAKE_BINARY_DIR}/java.stamp"
    DEPENDS ${JAVA_SOURCES} VERBATIM)
add_custom_target(java_classes ALL DEPENDS "${CMAKE_BINARY_DIR}/java.stamp")
add_library(brpc_echo_jni SHARED native/bridge.cpp)
add_dependencies(brpc_echo_jni java_classes)
target_include_directories(brpc_echo_jni PRIVATE "${CMAKE_BINARY_DIR}/generated-jni" "${JAVA_JNI_INCLUDE}" "${JAVA_JNI_PLATFORM}")
target_link_libraries(brpc_echo_jni PRIVATE echo_core)
target_link_options(brpc_echo_jni PRIVATE -Wl,-z,defs)

add_executable(echo_native_client native/native_client.cpp)
add_executable(echo_native_server native/native_server.cpp)
target_link_libraries(echo_native_client PRIVATE echo_core)
target_link_libraries(echo_native_server PRIVATE echo_core)
foreach(_target brpc_echo_jni echo_native_client echo_native_server)
    set_target_properties(${_target} PROPERTIES BUILD_RPATH "${_brpc_lib_dir}")
endforeach()

include(CTest)
if(BUILD_TESTING)
    add_executable(echo_statistics_test native/statistics_test.cpp)
    target_compile_features(echo_statistics_test PRIVATE cxx_std_14)
    add_test(NAME native_histogram COMMAND echo_statistics_test)
    add_test(NAME java_tcp_contract
        COMMAND "${Java_JAVA_EXECUTABLE}" -Xcheck:jni "-Djava.library.path=${CMAKE_BINARY_DIR}"
                -cp "${CMAKE_BINARY_DIR}/classes" prototype.echo.ContractTest)
    set_tests_properties(java_tcp_contract PROPERTIES TIMEOUT 45)
endif()

message(STATUS "Existing bRPC library: ${BRPC_LIBRARY}")
message(STATUS "Existing bRPC headers: ${BRPC_INCLUDE_DIR}")
message(STATUS "JDK: ${_java_home}; C++ standard: ${_cpp_standard}")
message(STATUS "URMA requested support: ${ECHO_ENABLE_URMA}; runtime device/data path NOT verified by this build")
